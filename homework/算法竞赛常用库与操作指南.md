**参考文件**: `homework/2026_4_24/maxway.cpp`

```cpp
// 解析 root = [1,2,3,null,null,15,7]
vector<string> parseInput(string line) {
    // 去掉空格
    string content;
    for (char c : line) {
        if (c != ' ') content += c;
    }
  
    // 提取方括号内容
    int l = content.find('[');
    int r = content.find(']');
    content = content.substr(l + 1, r - l - 1);
  
    // 按逗号分割
    vector<string> vals;
    string cur;
    for (char c : content) {
        if (c == ',') {
            vals.push_back(cur);
            cur.clear();
        } else {
            cur += c;
        }
    }
    if (!cur.empty()) vals.push_back(cur);
  
    return vals;
}
```

### 5.2 二叉树

**参考文件**: `homework/2026_4_24/maxway.cpp`

```cpp
struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
};

// 从层序数组建树
TreeNode* buildTree(vector<string>& vals) {
    if (vals.empty() || vals[0] == "null") return nullptr;
  
    TreeNode* root = new TreeNode(stoi(vals[0]));
    queue<TreeNode*> q;
    q.push(root);
  
    int i = 1;
    while (!q.empty() && i < vals.size()) {
        TreeNode* cur = q.front();
        q.pop();
  
        // 左子节点
        if (i < vals.size() && vals[i] != "null") {
            cur->left = new TreeNode(stoi(vals[i]));
            q.push(cur->left);
        }
        i++;
  
        // 右子节点
        if (i < vals.size() && vals[i] != "null") {
            cur->right = new TreeNode(stoi(vals[i]));
            q.push(cur->right);
        }
        i++;
    }
  
    return root;
}

// DFS遍历
void dfs(TreeNode* root) {
    if (!root) return;
    // 处理当前节点
    dfs(root->left);
    dfs(root->right);
}

// BFS遍历
void bfs(TreeNode* root) {
    if (!root) return;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        TreeNode* cur = q.front();
        q.pop();
        // 处理当前节点
        if (cur->left) q.push(cur->left);
        if (cur->right) q.push(cur->right);
    }
}
```
